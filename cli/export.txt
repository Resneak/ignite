function avs_accept() {
	// make sure they have selected one of the options
	var rbs = $('input[name=__avs_select]:checked');
	if (!rbs.val()) {
		alert('Please select an address to use.');
		return false;
	}
	// if they selected suggested instead of entered, then update the form before submitting
	if (rbs.val() == 1) {
		var resp = $('#avs_select_options').data('avsdata');
		var avsfields = ['address1','address2','city','state','postalCode','country'];
		for (var i in avsfields) {
			var fn = avsfields[i];
			switch (fn) {
				case 'address1':
					jQuery("#dwfrm_singleshipping_shippingAddress_addressFields_address1").val(resp.address_data.output[fn]);
					break;
				case 'address2':
					jQuery("#dwfrm_singleshipping_shippingAddress_addressFields_address2").val(resp.address_data.output[fn]);
					break;
				case 'city':
					jQuery("#dwfrm_singleshipping_shippingAddress_addressFields_city").val(resp.address_data.output[fn]);
					break;
				case 'state':
					jQuery("#dwfrm_singleshipping_shippingAddress_addressFields_states_state").val(resp.address_data.output[fn]);
					break;
				case 'postalCode':
					jQuery("#dwfrm_singleshipping_shippingAddress_addressFields_zip").val(resp.address_data.output[fn]);
					break;
				case 'country':
					jQuery("#dwfrm_singleshipping_shippingAddress_addressFields_country").val(resp.address_data.output[fn]);
					break;
			}
		}

	} else {
		$('#avs_select_options').data('confirmed-addr',true);
	}
	//$('#chkAddressRight').show();
	//console.log('submitting form');
	jQuery("#submitshippingbtn").click();
	return false;
}

		    t--;
		    return -c/2 * (t*(t-2) - 1) + b;

		        change = to - start,
		        increment = 20;
		    let currentTime = 0;

		        const val = Math.easeInOutQuad(currentTime, start, change, duration);
